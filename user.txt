<script setup lang="ts">
import { ref } from "vue";
import { useRouter, useRoute } from "vue-router";
import { useQuasar } from "quasar";

interface NavItem {
  name: string;
  label: string;
  icon: string;
  to: string;
}

interface CurrentUser {
  name: string;
  role: string;
  badge: string;
}

defineEmits<{ (e: "refresh"): void; (e: "logout"): void }>();

const router = useRouter();
const route = useRoute();
const $q = useQuasar();

const currentUser: CurrentUser = {
  name: "นพ.สมชาย ตั้งใจ (Admin Wellness)",
  role: "หัวหน้ากลุ่มงาน Wellness & ออกหน่วย",
  badge: "Admin Wellness"
};

const navItems: NavItem[] = [
  {
    name: "dashboard",
    label: "Dashboard สรุปผล",
    icon: "insights",
    to: "/dashboard"
  },
  {
    name: "trips",
    label: "การออกหน่วย (Pre/Post Trip)",
    icon: "event_available",
    to: "/trips"
  },
  {
    name: "expenses",
    label: "ติดตามเบิกจ่าย & แจ้งเตือน",
    icon: "receipt_long",
    to: "/expenses"
  },
  {
    name: "kits",
    label: "ชุดตรวจ & สิทธิ์การรักษา",
    icon: "medical_information",
    to: "/kits"
  },
  {
    name: "cost-config",
    label: "ตั้งค่าต้นทุน (Cost Config)",
    icon: "tune",
    to: "/cost-config"
  }
];

const activeTab = ref(
  navItems.find(n => route.path.startsWith(n.to))?.name ?? navItems[0].name
);

// Drawer starts closed. On desktop, `show-if-above` forces it visible
// regardless of this value. On mobile it stays hidden until the user
// taps the menu button, then behaves as an overlay.
const leftDrawerOpen = ref(false);

function goTo(item: NavItem) {
  activeTab.value = item.name;
  router.push(item.to);

  // On mobile the drawer is an overlay sitting on top of the page,
  // so close it after picking a destination. On desktop it's
  // persistent and this has no visual effect either way.
  if ($q.screen.lt.md) {
    leftDrawerOpen.value = false;
  }
}
</script>
